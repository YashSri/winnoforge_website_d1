[ANIM-BG-01]: Smooth Cursor Lerp Spotlight (Ambient Damped Follower)
--------------------------------------------------------------------------------
- Used In FORGE:
  components/shared/AmbientBackground.tsx (Lines 15-51, 58-62)
- Visual Description:
  A massive, soft 500px glowing radial gradient orb that tracks the user's cursor across
  the screen with ultra-smooth 60fps linear damping (lerp). The orb never jumps abruptly;
  it smoothly glides and catches up with a subtle inertia delay, illuminating background
  glass panels and text with a premium blue aura.
- Tech Stack & Dependencies:
  React (useEffect, useRef, requestAnimationFrame), Vanilla CSS (will-change-transform, blur-[100px])
- Exact Implementation Code:
  ```tsx
  const mouseGlowRef = useRef<HTMLDivElement>(null);
  const targetPos = useRef({ x: -1000, y: -1000 });
  const currentPos = useRef({ x: -1000, y: -1000 });

  useEffect(() => {
    let animId: number;
    const handleMouseMove = (e: MouseEvent) => {
      targetPos.current = { x: e.clientX, y: e.clientY };
    };

    const updatePosition = () => {
      // Damping factor: 0.08 produces an elegant, fluid glide
      currentPos.current.x += (targetPos.current.x - currentPos.current.x) * 0.08;
      currentPos.current.y += (targetPos.current.y - currentPos.current.y) * 0.08;

      if (mouseGlowRef.current) {
        mouseGlowRef.current.style.transform = `translate3d(${currentPos.current.x - 250}px, ${currentPos.current.y - 250}px, 0)`;
      }
      animId = requestAnimationFrame(updatePosition);
    };

    window.addEventListener("mousemove", handleMouseMove, { passive: true });
    animId = requestAnimationFrame(updatePosition);
    return () => {
      window.removeEventListener("mousemove", handleMouseMove);
      cancelAnimationFrame(animId);
    };
  }, []);

  // JSX:
  <div
    ref={mouseGlowRef}
    className="will-change-transform pointer-events-none fixed top-0 left-0 w-[500px] h-[500px] rounded-full bg-gradient-to-r from-[#1683E8]/12 via-[#00B4D8]/08 to-transparent blur-[100px]"
  />
  ```

- Used In FORGE:

- Visual Description:
  A clean, zero-dependency lightweight React wrapper component using IntersectionObserver.
  Any element wrapped with `<ScrollReveal delay={150}>` remains hidden until 8% visible,
  then smoothly transitions with opacity: 1 and translateY(0) using cubic-bezier(0.22, 1, 0.36, 1).
- Tech Stack & Dependencies:
  React, IntersectionObserver API, prefers-reduced-motion media query
- Exact Implementation Code:
  ```tsx
  export default function ScrollReveal({ children, className = "", delay = 0 }) {
    const ref = useRef<HTMLDivElement>(null);

    useEffect(() => {
      const el = ref.current;
      if (!el) return;
      if (window.matchMedia("(prefers-reduced-motion: reduce)").matches) {
        el.style.opacity = "1";
        el.style.transform = "none";
        return;
      }
      const observer = new IntersectionObserver(([entry]) => {
        if (entry.isIntersecting) {
          el.style.opacity = "1";
          el.style.transform = "translateY(0)";
          observer.unobserve(el);
        }
      }, { threshold: 0.08 });
      observer.observe(el);
      return () => observer.disconnect();
    }, []);

    return (
      <div
        ref={ref}
        className={className}
        style={{
          opacity: 0,
          transform: "translateY(24px)",
          transition: `opacity 0.65s cubic-bezier(0.22, 1, 0.36, 1) ${delay}ms, transform 0.65s cubic-bezier(0.22, 1, 0.36, 1) ${delay}ms`,
        }}
      >
        {children}
      </div>
    );
  }
  ```
